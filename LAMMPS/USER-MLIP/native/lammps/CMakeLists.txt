cmake_minimum_required(VERSION 3.22)
project(gpml_native_lammps LANGUAGES CXX)

set(LAMMPS_SOURCE_DIR "" CACHE PATH "Upstream LAMMPS root or src directory")
if(EXISTS "${LAMMPS_SOURCE_DIR}/lammps.h")
  get_filename_component(_lammps_root "${LAMMPS_SOURCE_DIR}/.." ABSOLUTE)
else()
  get_filename_component(_lammps_root "${LAMMPS_SOURCE_DIR}" ABSOLUTE)
endif()
if(NOT EXISTS "${_lammps_root}/cmake/CMakeLists.txt" OR NOT EXISTS "${_lammps_root}/src/lammps.h")
  message(FATAL_ERROR "LAMMPS_SOURCE_DIR must identify an upstream LAMMPS root or src directory")
endif()

set(IMRMLP_NATIVE_BACKEND CPU CACHE STRING "IMR-MLP backend to link: CPU, CUDA, or HIP")
set_property(CACHE IMRMLP_NATIVE_BACKEND PROPERTY STRINGS CPU CUDA HIP)
string(TOUPPER "${IMRMLP_NATIVE_BACKEND}" IMRMLP_NATIVE_BACKEND)
if(NOT IMRMLP_NATIVE_BACKEND MATCHES "^(CPU|CUDA|HIP)$")
  message(FATAL_ERROR "IMRMLP_NATIVE_BACKEND must be CPU, CUDA, or HIP")
endif()
if(DEFINED BUILD_MPI AND NOT BUILD_MPI)
  message(FATAL_ERROR "mlip/native requires BUILD_MPI=ON to match the MPI IMR-MLP library")
endif()
set(BUILD_MPI ON CACHE BOOL "Build LAMMPS with MPI")
set(LAMMPS_HEADER_DIR "${_lammps_root}/src")
get_filename_component(_native "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
get_filename_component(_root "${_native}/../../.." ABSOLUTE)
if(IMRMLP_NATIVE_BACKEND STREQUAL "CUDA")
  set(_mode mpi-cuda)
  set(_library libimrmlp-mpi.so)
elseif(IMRMLP_NATIVE_BACKEND STREQUAL "HIP")
  set(_mode mpi-hip)
  set(_library libimrmlp-mpi.so)
else()
  set(_mode mpi)
  set(_library libimrmlp-mpi.so)
endif()
set(IMRMLP_LIBRARY "${_root}/build/${_mode}/${_library}" CACHE FILEPATH "Matching MPI IMR-MLP library")
set(IMRMLP_FLAGS_FILE "${_root}/build/${_mode}/.flags" CACHE FILEPATH "Compile flags of that IMR-MLP library")
include("${_native}/cmake/IMRMLPNative.cmake")
if(NOT IMRMLP_NATIVE_BACKEND STREQUAL IMRMLP_DETECTED_BACKEND)
  message(FATAL_ERROR "IMRMLP_NATIVE_BACKEND=${IMRMLP_NATIVE_BACKEND} conflicts with ${IMRMLP_DETECTED_BACKEND} IMRMLP_FLAGS_FILE")
endif()

# The GPU kernels are already in IMR-MLP. This entry point only compiles C++,
# including for CUDA/HIP, and does not probe an unrelated device compiler.
add_subdirectory("${_lammps_root}/cmake" upstream)
if(NOT TARGET lammps OR NOT TARGET lmp OR NOT COMMAND AddStyleHeader OR NOT COMMAND GenerateStyleHeader)
  message(FATAL_ERROR "This LAMMPS CMake version lacks the required lammps/lmp targets or style-header helpers")
endif()
foreach(kind PAIR COMPUTE)
  get_property(_headers GLOBAL PROPERTY ${kind})
  foreach(_header IN LISTS _headers)
    file(READ "${_header}" _contents)
    if(_contents MATCHES "(PairStyle|ComputeStyle)[ \t\r\n]*\\([ \t\r\n]*mlip/native([ \t\r\n]*,|/charge[ \t\r\n]*,|/latent[ \t\r\n]*,)")
      message(FATAL_ERROR "Duplicate native style in ${_header}; use a clean LAMMPS source tree without an installed mlip/native copy")
    endif()
  endforeach()
endforeach()
foreach(source pair_mlip_native compute_mlip_native)
  target_sources(lammps PRIVATE "${_native}/${source}.cpp")
  # IMR-MLP's ABI macros and forced precompiled header belong only to the native
  # translation units, not upstream LAMMPS or its optional packages.
  set_property(SOURCE "${_native}/${source}.cpp" TARGET_DIRECTORY lammps APPEND PROPERTY COMPILE_OPTIONS ${IMRMLP_ABI_FLAGS})
  set_property(SOURCE "${_native}/${source}.cpp" TARGET_DIRECTORY lammps APPEND PROPERTY INCLUDE_DIRECTORIES "${IMRMLP_ROOT}" "${IMRMLP_ROOT}/deplibs")
endforeach()
target_compile_features(lammps PRIVATE cxx_std_23)
target_include_directories(lammps PRIVATE "${_native}")
target_link_libraries(lammps PRIVATE MPI::MPI_CXX Eigen3::Eigen "${IMRMLP_LIBRARY}")
AddStyleHeader("${_native}/pair_mlip_native.h" PAIR)
AddStyleHeader("${_native}/compute_mlip_native.h" COMPUTE)
get_target_property(_lammps_binary lammps BINARY_DIR)
GenerateStyleHeader("${_lammps_binary}/styles" PAIR pair)
GenerateStyleHeader("${_lammps_binary}/styles" COMPUTE compute)
foreach(target lammps lmp)
  set_property(TARGET ${target} APPEND PROPERTY BUILD_RPATH "${IMRMLP_LIBRARY_DIR}")
  set_property(TARGET ${target} APPEND PROPERTY INSTALL_RPATH "${IMRMLP_LIBRARY_DIR}")
endforeach()
set_target_properties(lmp PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
message(STATUS "Native styles embedded in standard lmp: mlip/native, mlip/native/charge, mlip/native/latent (${IMRMLP_NATIVE_BACKEND})")

include(CTest)
include("${_native}/cmake/NativeTests.cmake")
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  native_add_tests("$<TARGET_FILE:lmp>" "")
  add_test(NAME native_lammps_build_contract COMMAND "${Python3_EXECUTABLE}"
    "${_native}/tests/test_lammps_build.py" --lammps-source "${_lammps_root}" --imrmlp-library "${IMRMLP_LIBRARY}")
  set_tests_properties(native_lammps_build_contract PROPERTIES TIMEOUT 300)
endif()
