cmake_minimum_required(VERSION 3.18)
project(mlip_native_plugin LANGUAGES CXX)

set(LAMMPS_HEADER_DIR "" CACHE PATH "Source headers of the target LAMMPS executable")
if(NOT EXISTS "${LAMMPS_HEADER_DIR}/lammpsplugin.h")
  message(FATAL_ERROR "LAMMPS_HEADER_DIR must contain lammpsplugin.h")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/IMRMLPNative.cmake")

add_library(mlip_native_plugin MODULE native_plugin.cpp pair_mlip_native.cpp compute_mlip_native.cpp)
native_target(mlip_native_plugin)
set_target_properties(mlip_native_plugin PROPERTIES PREFIX "" SUFFIX ".so")
if(UNIX AND NOT APPLE)
  target_link_options(mlip_native_plugin PRIVATE -rdynamic)
elseif(APPLE)
  target_link_options(mlip_native_plugin PRIVATE -Wl,-undefined,dynamic_lookup)
endif()

include(CTest)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/NativeTests.cmake")
set(LAMMPS_EXECUTABLE "" CACHE FILEPATH "MPI LAMMPS executable with PLUGIN support")
if(BUILD_TESTING)
  native_add_tests("${LAMMPS_EXECUTABLE}" "$<TARGET_FILE:mlip_native_plugin>")
endif()
