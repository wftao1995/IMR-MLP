cmake_minimum_required(VERSION 3.22)
project(mtp_standalone VERSION 1.0 LANGUAGES CXX)

set(MTP_BACKEND CPU CACHE STRING "Execution backend: CPU, CUDA or HIP")
set_property(CACHE MTP_BACKEND PROPERTY STRINGS CPU CUDA HIP)
string(TOUPPER "${MTP_BACKEND}" MTP_BACKEND)
if(NOT MTP_BACKEND MATCHES "^(CPU|CUDA|HIP)$")
  message(FATAL_ERROR "MTP_BACKEND must be CPU, CUDA or HIP")
endif()
option(MTP_OPENMP "Build mlip/opt/omp (target LAMMPS needs OPENMP)" OFF)
option(MTP_BUILD_ADAPTERS "Build the LAMMPS plugin and static adapter" ON)
option(MTP_BUILD_TESTS "Build standalone core tests" ON)
option(MTP_BUILD_MPI_TESTS "Build collective error propagation tests" ${MTP_BUILD_ADAPTERS})
option(MTP_BUILD_GPU_TESTS "Build device-required core regression tests (Catch2 3)" OFF)
set(LAMMPS_SOURCE_DIR "" CACHE PATH "Target LAMMPS root or src directory")
set(LAMMPS_LIBRARY "" CACHE FILEPATH "Optional liblammps for the static test runner")
set(LAMMPS_EXTRA_LIBRARIES "" CACHE STRING "Extra link libraries required by a static liblammps")
set(MTP_LAMMPS_DEFINITIONS "LAMMPS_SMALLBIG" CACHE STRING "ABI definitions of target LAMMPS")
set(MTP_ARCH_FLAGS "" CACHE STRING "Optional CPU architecture flags; no fast-math")
get_filename_component(MTP_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
if(EXISTS "${MTP_SOURCE_ROOT}/mlip-core/mlip.cpp")
  set(MTP_CORE "${MTP_SOURCE_ROOT}/mlip-core")
else()
  get_filename_component(MTP_CORE "${MTP_SOURCE_ROOT}/../../runtime/mtp" ABSOLUTE)
endif()

# The exported package contains the shared format and metadata headers. An in-checkout build copies
# exactly the same headers; no GPML include directory enters the runtime target.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/mtp_metadata.h")
  set(_metadata "${CMAKE_CURRENT_SOURCE_DIR}/vendor")
else()
  get_filename_component(_metadata "${MTP_SOURCE_ROOT}/../../tools" ABSOLUTE)
endif()
foreach(_header mtp_metadata.h picojson.h gpm_container.h gpm_mtp.h)
  if(NOT EXISTS "${_metadata}/${_header}")
    message(FATAL_ERROR "Missing standalone metadata header ${_metadata}/${_header}")
  endif()
  configure_file("${_metadata}/${_header}" "${CMAKE_CURRENT_BINARY_DIR}/metadata/${_header}" COPYONLY)
endforeach()

add_library(mtp_runtime STATIC "${MTP_CORE}/mlip.cpp" "${MTP_CORE}/cpu/mtp_cpu.cpp" "${MTP_CORE}/radial_basis_setup.cpp")
add_library(MTP::runtime ALIAS mtp_runtime)
option(MTP_ZSTD "Read zstd-compressed GPM blocks" OFF)
if(MTP_ZSTD)
  find_path(MTP_ZSTD_INCLUDE zstd.h REQUIRED)
  find_library(MTP_ZSTD_LIBRARY zstd REQUIRED)
  target_compile_definitions(mtp_runtime PUBLIC USE_ZSTD)
  target_include_directories(mtp_runtime PUBLIC "${MTP_ZSTD_INCLUDE}")
  target_link_libraries(mtp_runtime PUBLIC "${MTP_ZSTD_LIBRARY}")
endif()

# Bundled, pinned headers are private to host-side matrix initialization.
set_source_files_properties("${MTP_CORE}/radial_basis_setup.cpp" PROPERTIES
  INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/vendor/eigen3")
target_compile_features(mtp_runtime PUBLIC cxx_std_17)
target_include_directories(mtp_runtime PUBLIC "${MTP_SOURCE_ROOT}" "${MTP_CORE}"
  "${CMAKE_CURRENT_BINARY_DIR}/metadata")
set_target_properties(mtp_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang|IntelLLVM")
  target_compile_options(mtp_runtime PRIVATE -fopenmp-simd)
endif()
if(MTP_ARCH_FLAGS)
  separate_arguments(_arch NATIVE_COMMAND "${MTP_ARCH_FLAGS}")
  target_compile_options(mtp_runtime PRIVATE ${_arch})
endif()

# No GPU compiler/toolkit probe occurs in the CPU branch.
if(NOT MTP_BACKEND STREQUAL "CPU")
  if(MTP_BACKEND STREQUAL "CUDA")
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
  else()
    enable_language(HIP)
    find_package(hip REQUIRED CONFIG)
    set_source_files_properties("${MTP_CORE}/gpu-direct/mtp_gpu_direct.cu"
      "${MTP_CORE}/gpu-direct/sog_gpu.cu" PROPERTIES LANGUAGE HIP)
  endif()
  add_library(mtp_gpu STATIC "${MTP_CORE}/gpu-direct/mtp_gpu_direct.cu"
    "${MTP_CORE}/gpu-direct/sog_gpu.cu")
  target_include_directories(mtp_gpu PUBLIC "${MTP_SOURCE_ROOT}" "${MTP_CORE}")
  target_compile_definitions(mtp_gpu PUBLIC IMRMLP_GPU IMRMLP_${MTP_BACKEND})
  target_link_libraries(mtp_gpu PUBLIC mtp_runtime)
  set_target_properties(mtp_gpu PROPERTIES CXX_STANDARD 17 CUDA_STANDARD 17
    HIP_STANDARD 17 CXX_STANDARD_REQUIRED ON CUDA_STANDARD_REQUIRED ON
    HIP_STANDARD_REQUIRED ON POSITION_INDEPENDENT_CODE ON)
  if(MTP_BACKEND STREQUAL "CUDA")
    target_link_libraries(mtp_gpu PUBLIC CUDA::cudart)
    target_compile_options(mtp_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  else()
    target_link_libraries(mtp_gpu PUBLIC hip::host)
  endif()
endif()

if(MTP_BUILD_ADAPTERS)
  if(EXISTS "${LAMMPS_SOURCE_DIR}/src/lammps.h")
    set(_lmp_headers "${LAMMPS_SOURCE_DIR}/src")
  else()
    set(_lmp_headers "${LAMMPS_SOURCE_DIR}")
  endif()
  if(NOT EXISTS "${_lmp_headers}/lammpsplugin.h")
    message(FATAL_ERROR "LAMMPS_SOURCE_DIR must identify matching LAMMPS source headers; or set MTP_BUILD_ADAPTERS=OFF")
  endif()
  find_package(MPI REQUIRED COMPONENTS CXX)
  add_library(mtp_lammps_adapters STATIC register.cpp "${MTP_SOURCE_ROOT}/opt/pair_MLIP_opt.cpp")
  add_library(MTP::lammps_adapters ALIAS mtp_lammps_adapters)
  set_target_properties(mtp_lammps_adapters PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_compile_features(mtp_lammps_adapters PUBLIC cxx_std_17)
  target_include_directories(mtp_lammps_adapters PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}"
    "${_lmp_headers}" "${MTP_SOURCE_ROOT}/opt")
  target_compile_definitions(mtp_lammps_adapters PUBLIC ${MTP_LAMMPS_DEFINITIONS})
  target_link_libraries(mtp_lammps_adapters PUBLIC mtp_runtime MPI::MPI_CXX)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang|IntelLLVM")
    target_compile_options(mtp_lammps_adapters PRIVATE -fopenmp-simd)
  endif()
  if(MTP_ARCH_FLAGS)
    target_compile_options(mtp_lammps_adapters PRIVATE ${_arch})
  endif()
  if(MTP_OPENMP)
    find_package(OpenMP REQUIRED COMPONENTS CXX)
    target_sources(mtp_lammps_adapters PRIVATE "${MTP_SOURCE_ROOT}/opt/pair_MLIP_omp.cpp")
    target_include_directories(mtp_lammps_adapters PUBLIC "${_lmp_headers}/OPENMP")
    target_compile_definitions(mtp_lammps_adapters PUBLIC MTP_STANDALONE_OPENMP)
    target_link_libraries(mtp_lammps_adapters PUBLIC OpenMP::OpenMP_CXX)
  endif()
  if(TARGET mtp_gpu)
    target_sources(mtp_lammps_adapters PRIVATE "${MTP_SOURCE_ROOT}/pair_MLIP_gpu_direct.cpp")
    target_compile_definitions(mtp_lammps_adapters PUBLIC MTP_STANDALONE_GPU)
    target_link_libraries(mtp_lammps_adapters PUBLIC mtp_gpu)
  endif()
  add_library(mtp_standalone_plugin MODULE plugin.cpp)
  target_link_libraries(mtp_standalone_plugin PRIVATE mtp_lammps_adapters)
  set_target_properties(mtp_standalone_plugin PROPERTIES PREFIX "")
  if(APPLE)
    target_link_options(mtp_standalone_plugin PRIVATE -Wl,-undefined,dynamic_lookup)
  endif()
  if(LAMMPS_LIBRARY)
    add_executable(mtp_lammps tests/lammps_runner.cpp)
    target_link_libraries(mtp_lammps PRIVATE mtp_lammps_adapters "${LAMMPS_LIBRARY}"
      ${LAMMPS_EXTRA_LIBRARIES} ${CMAKE_DL_LIBS})
  endif()
  install(TARGETS mtp_standalone_plugin LIBRARY DESTINATION lib)
endif()

if(MTP_BUILD_TESTS)
  enable_testing()
  if(MTP_BUILD_GPU_TESTS)
    if(NOT TARGET mtp_gpu)
      message(FATAL_ERROR "MTP_BUILD_GPU_TESTS requires MTP_BACKEND=CUDA or HIP")
    endif()
    find_package(Catch2 3 REQUIRED)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_standalone_gpu.cpp")
      set(_gpu_test "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_standalone_gpu.cpp")
    else()
      get_filename_component(_gpu_test "${MTP_SOURCE_ROOT}/../../unitest/ut_standalone_gpu.cpp" ABSOLUTE)
    endif()
    add_executable(test_standalone_gpu "${_gpu_test}")
    target_link_libraries(test_standalone_gpu PRIVATE mtp_gpu Catch2::Catch2WithMain)
    add_test(NAME mtp_gpu COMMAND test_standalone_gpu)
    set_tests_properties(mtp_gpu PROPERTIES TIMEOUT 300 LABELS gpu
      ENVIRONMENT "IMRMLP_REQUIRE_GPU=1")
  endif()
  if(MTP_BUILD_MPI_TESTS)
    find_package(MPI REQUIRED COMPONENTS CXX)
    add_executable(test_model_mpi tests/test_model_mpi.cpp)
    target_link_libraries(test_model_mpi PRIVATE mtp_runtime MPI::MPI_CXX)
    foreach(_ranks 2 4)
      add_test(NAME mtp_mpi_errors_${_ranks} COMMAND ${MPIEXEC_EXECUTABLE}
        ${MPIEXEC_NUMPROC_FLAG} ${_ranks} ${MPIEXEC_PREFLAGS}
        $<TARGET_FILE:test_model_mpi> ${MPIEXEC_POSTFLAGS}
        "${CMAKE_CURRENT_BINARY_DIR}/missing-mtp-model-for-error-test.mtp")
      set_tests_properties(mtp_mpi_errors_${_ranks} PROPERTIES
        TIMEOUT 30 PROCESSORS ${_ranks} LABELS mpi)
    endforeach()
  endif()
  add_executable(test_radial tests/test_radial.cpp)
  target_link_libraries(test_radial PRIVATE mtp_runtime)
  add_test(NAME mtp_radial COMMAND test_radial)
  add_executable(mtp_model_probe tests/model_probe.cpp)
  target_link_libraries(mtp_model_probe PRIVATE mtp_runtime)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME mtp_reader COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_reader.py" --probe "$<TARGET_FILE:mtp_model_probe>")
  endif()
  if(EXISTS "${MTP_CORE}/cpu/tests/test_mtp_cpu.cpp")
    add_executable(test_mtp_cpu "${MTP_CORE}/cpu/tests/test_mtp_cpu.cpp")
    target_link_libraries(test_mtp_cpu PRIVATE mtp_runtime)
    if(MTP_OPENMP)
      find_package(OpenMP REQUIRED COMPONENTS CXX)
      target_link_libraries(test_mtp_cpu PRIVATE OpenMP::OpenMP_CXX)
    endif()
    add_test(NAME mtp_cpu COMMAND test_mtp_cpu)
  endif()
endif()
install(TARGETS mtp_runtime ARCHIVE DESTINATION lib)
